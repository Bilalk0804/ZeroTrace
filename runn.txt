# Navigate to your project directory
cd /path/to/ZeroTrace

# Build using the Makefile
make build

# Or build manually
go build -o build/zerotrace nwipe_main.go config.go utils.go


# Run the built binary
sudo ./build/zerotrace

# Or install system-wide first
make install
sudo zerotrace


# Make the test script executable
chmod +x test_install.sh

# Run the test script
./test_install.sh


# Install Python dependencies
pip install -r requirements.txt

# Build Go backend (Linux only)
make build

# Run integrated GUI
python3 gui_integration.py
